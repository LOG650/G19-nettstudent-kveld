% Global takhøyde for figurer. Pandoc-spesifisert width=NN% beholdes,
% men keepaspectratio garanterer proporsjonal skalering, og max
% totalheight stopper figuren i å renne over sidekanten dersom pandoc
% emitterer height=\textheight (som er default i pandoc 3.x).
\setkeys{Gin}{keepaspectratio,max width=\linewidth,max totalheight=0.8\textheight}

% Slakkere LaTeX-float-parametre slik at figurer lettere finner en
% gyldig plassering når siden allerede er full av tekst og tabeller.
% Default-verdiene fører ellers til Overfull \vbox når en moderat
% høy figur (≈9 cm) presses inn nederst på en side med liten ledig
% plass i stedet for å flyte til neste side.
\renewcommand{\topfraction}{0.9}
\renewcommand{\bottomfraction}{0.9}
\renewcommand{\textfraction}{0.08}
\renewcommand{\floatpagefraction}{0.7}

% Fjern 'h' fra default float-placement (pandoc setter htbp). Med 'h'
% kan LaTeX presse en figur inline selv når plassen er marginal, slik
% at figurens nedre kant klippes av sidemargen. Med kun 'tbp' tvinges
% figurer til å flyte til topp/bunn/floatside — standard akademisk
% rapportkonvensjon.
\makeatletter
\renewcommand{\fps@figure}{tbp}
\renewcommand{\fps@table}{tbp}
\makeatother

\setlength{\tabcolsep}{4pt}
\renewcommand{\arraystretch}{1.1}
\AtBeginEnvironment{longtable}{\small}

% Caption-styling: ingen auto-prefiks (vi har manuell "Figur X.Y" /
% "Tabell X.Y" i caption-teksten), liten italic skrift sentrert under
% figur og tabell.
\captionsetup[figure]{labelformat=empty, font={small,it}, justification=centering}
\captionsetup[longtable]{labelformat=empty, font={small,it}, justification=centering}

% Tillat linjebryting i lange ubrudte strenger (klassenavn, parametre osv.)
% slik at de ikke renner over kolonnegrensen i smale tabellceller. Visuelt
% nøytralt når strengen får plass uten bryting.
\newcommand{\splitcode}[1]{{\ttfamily\seqsplit{#1}}}
\newcommand{\splitword}[1]{\seqsplit{#1}}

% Default-fonten Latin Modern har ikke disse Unicode-symbolene; mapp dem
% til pifont-glyfer / LaTeX-ekvivalenter.
\newunicodechar{☐}{\ding{110}}
\newunicodechar{☒}{\ding{55}}
\newunicodechar{−}{\ensuremath{-}}
